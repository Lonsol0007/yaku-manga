# Desktop-only build dependencies for producing translation packs.
# Nothing here ships in the APK - the phone only ever sees the exported .onnx and .json files.

torch>=2.2
onnx>=1.16
onnxruntime>=1.18
# torch 2.9+ routes torch.onnx.export through the dynamo exporter, which imports onnxscript
# lazily and is not declared as a torch dependency - without it the detector export dies with
# ModuleNotFoundError on the first call.
onnxscript>=0.7
sentencepiece>=0.2
python-doctr[torch]>=0.8

# Optimum 2.x moved the ONNX exporter out into its own distribution; installing plain
# `optimum[exporters]` gets you a package with no optimum.exporters.onnx module at all.
# optimum-onnx pins a compatible transformers, so transformers is deliberately unpinned here.
optimum>=2,<3
optimum-onnx>=0.1

# manga-ocr's tokenizer is BertJapanese, which needs a Japanese morphological analyser
fugashi>=1.3
unidic-lite>=1.0.8
